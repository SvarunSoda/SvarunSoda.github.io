111{
112 uint32_t startTimeMicros;
113 String data;

115 // buffers for duty cycle measurements...
116 uint32_t timestampBuff[OscillatorNum][iters];
117 uint32_t cycleBuff[OscillatorNum][2][iters];

119 
120 Serial.println("running frequency-locking loop:\n" + FLLToString(iters, batches, operatingCycles, operatingCyclesTolerance, rollingAverage) + "\n...");
121 
122 // executing for each FLL loop...
123 for (uint32_t j = 0; j < batches; j++)
124 {
125 //Serial.println("\n-- FLL BATCH #" + String(j + 1) + " --");
126 // running frequency-locking closed-loop for a pre-determined amount of iterations...
127 startTimeMicros = micros();
128 for (uint32_t k = 0; k < iters; k++)

130 // iterating over each oscillator...

170// function that executes a single frequency-locking closed-loop iteration for a single oscillator
171inline void FLLRunIter(const uint8_t oscillatorIdx, const uint32_t iter, const uint32_t startTimeMicros, const uint32_t operatingCycles, const uint32_t operatingCyclesTolerance, const uint32_t rollingAverage, uint32_t* timestampBuff, uint32_t* cycleBuffHigh, bool* pauseBuffHigh, uint32_t* cycleBuffLow, bool* pauseBuffLow)
172{
173 uint32_t measuredCycles;
174 
175 // if the operating target cycles is 0, we don't run the iteration for the current oscillator...
176 if (operatingCycles == 0)
177 return;
178 
179 if (timestampBuff != nullptr)
180 timestampBuff[iter] = micros() - startTimeMicros;
181 
182 // measuring & adjusting the current oscillator's high duty cycle duration...
183 {
184 measuredCycles = OscillatorGetCyclesHigh(oscillatorIdx, operatingCycles, rollingAverage, true, iter, cycleBuffHigh, pauseBuffHigh);
185 
186 if (UnsignedGetDifference(measuredCycles, operatingCycles) > operatingCyclesTolerance)
187 {
188 if (measuredCycles < operatingCycles)
189 OscillatorHighDutyIncrease(oscillatorIdx, 1);
190 else if (measuredCycles > operatingCycles)
191 OscillatorHighDutyDecrease(oscillatorIdx, 1);
192 }
193 }
194 
195 // measuring & adjusting the current oscillator's low duty cycle duration...
196 {
197 measuredCycles = OscillatorGetCyclesLow(oscillatorIdx, operatingCycles, rollingAverage, true, iter, cycleBuffLow, pauseBuffLow);
198 
199 if (UnsignedGetDifference(measuredCycles, operatingCycles) > operatingCyclesTolerance)
200 {
201 if (measuredCycles < operatingCycles)
202 OscillatorLowDutyIncrease(oscillatorIdx, 1);
203 else if (measuredCycles > operatingCycles)
204 OscillatorLowDutyDecrease(oscillatorIdx, 1);
205 }
206 }
207}

231// function that measures a single duty cycle duration for a given oscillator
232FASTRUN __attribute__((noinline)) uint32_t OscillatorGetCycles(const uint8_t oscillatorIdx, const uint32_t operatingCycles, const bool invert, const uint32_t rollingAverage, const bool subtractPauses, const uint32_t buffIdx, uint32_t* cycleBuff, bool* pauseBuff)
233{
234 String str;
235 uint32_t startCycles, elapsedCycles, measuredCycles, averaged;
236 volatile bool checkState, prevCheckState;
237 bool pauseStarted, pauseEncountered;
238 
239 measuredCycles = 0;
240 averaged = 0;
241 pauseEncountered = false;
242 
243 // measuring the duty cycle for number of rolling averages...
244 for (uint32_t i = 0; i < rollingAverage; i++)
245 {
246 pauseStarted = false;
247 
248 // waiting for rising edge...
249 checkState = OscillatorGetState(oscillatorIdx, invert);
250 do
251 {
252 prevCheckState = checkState;
253 checkState = OscillatorGetState(oscillatorIdx, invert);
254 } while (!(checkState && (prevCheckState != checkState)));
255 
256 // waiting for falling edge, while measuring cycles...
257 startCycles = CPUCyclesGet();
258 checkState = OscillatorGetState(oscillatorIdx, invert);
259 do
260 {
261 // if we are subtracting pauses, and we detect an active pausing action on the oscillator that's being measured, we skip this oscillator...
262 if (subtractPauses)
263 {
264 if (OscillatorGetPause(oscillatorIdx))
265 {
266 pauseStarted = true;
267 break;
268 }
269 }
270 
271 prevCheckState = checkState;
272 checkState = OscillatorGetState(oscillatorIdx, invert);
273 } while (!(!checkState && (prevCheckState != checkState)));
274 
275 if (pauseStarted)
276 {
277 pauseEncountered = true;
278 continue;
279 }
280 
281 // measuring the elapsed cycles...
282 elapsedCycles = CPUCyclesGet() - startCycles;
283 if (averaged == 0)
284 measuredCycles = elapsedCycles;
285 else
286 measuredCycles = (measuredCycles + elapsedCycles) / 2;
287 averaged++;
288 }
289 
290 if (cycleBuff != nullptr)
291 cycleBuff[buffIdx] = measuredCycles;
292 if (pauseBuff != nullptr)
293 pauseBuff[buffIdx] = pauseEncountered;
294 
295 return measuredCycles;
296}

1472// function that measures a single duty cycle duration for a given oscillator
1473FASTRUN __attribute__((noinline)) uint32_t OscillatorGetCycles(const uint8_t oscillatorIdx, const bool invert, const uint32_t rollingAverage)
1474{
1475 uint32_t startCycles, elapsedCycles, measuredCycles;
1476 volatile bool checkState, prevCheckState;
1477 
1478 measuredCycles = 0;
1479 
1480 // measuring the duty cycle for number of rolling averages...
1481 for (uint32_t i = 0; i < rollingAverage; i++)
1482 {
1483 // waiting for rising edge...
1484 checkState = OscillatorGetState(oscillatorIdx, invert);
1485 do
1486 {
1487 prevCheckState = checkState;
1488 checkState = OscillatorGetState(oscillatorIdx, invert);
1489 } while (!(checkState && (prevCheckState != checkState)));
1490 
1491 // waiting for falling edge, while measuring cycles...
1492 startCycles = CPUCyclesGet();
1493 checkState = OscillatorGetState(oscillatorIdx, invert);
1494 do
1495 {
1496 prevCheckState = checkState;
1497 checkState = OscillatorGetState(oscillatorIdx, invert);
1498 } while (!(!checkState && (prevCheckState != checkState)));
1499 
1500 // saving the measured cycles...
1501 elapsedCycles = CPUCyclesGet() - startCycles;
1502 if (i == 0)
1503 measuredCycles = elapsedCycles;
1504 else
1505 measuredCycles = (measuredCycles + elapsedCycles) / 2;
1506 }
1507 
1508 return measuredCycles;
1509}

1600// saves data to a specified file on an SD card
1601inline void OscillatorTrackFrequenciesAll(const uint32_t operatingCycles, const uint32_t iters, const uint32_t startTimeMicros, const uint32_t rollingAverage, const String logFilePath)
1602{
1603 String data;
1604

1606 uint32_t timestampsBuff[iters];
1607 uint32_t cyclesHighBuff[iters][OscillatorNum];
1608 uint32_t cyclesLowBuff[iters][OscillatorNum];
1609 
1610 // tracking all oscillator operating duty cycle durations over a number of iterations...
1611 for (uint32_t i = 0; i < iters; i++)
1612 {
1613 timestampsBuff[i] = micros() - startTimeMicros;
1614 
1615 for (uint8_t j = 0; j < OscillatorNum; j++)
1616 {
1617 cyclesHighBuff[i][j] = OscillatorGetCycles(j, false, rollingAverage);
1618 cyclesLowBuff[i][j] = OscillatorGetCycles(j, true, rollingAverage);
1619 }
1620 }
1621 
1622 // saving measured duty cycle data...
1623 {
1624 SDCardInit();
1625 SDLogFileNew(logFilePath);
1626 for (uint32_t i = 0; i < iters; i++)
1627 {
1628 for (uint8_t j = 0; j < OscillatorNum; j++)
1629 {
1630 data = String(operatingCycles) + "|" +
1631 String(j) + "|0|0|" +
1632 String(i) + "|" +
1633 String(timestampsBuff[i]) + "|" +
1634 String(cyclesHighBuff[i][j]);

1636 
1637 data = String(operatingCycles) + "|" +
1638 String(j) + "|1|0|" +
1639 String(i) + "|" +
1640 String(timestampsBuff[i]) + "|" +
1641 String(cyclesLowBuff[i][j]);

1647 }
