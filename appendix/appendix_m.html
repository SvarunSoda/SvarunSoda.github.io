43def PlotDutyVsTime(data: dict, show: bool, savePath: str) -> None:
44 for targetDuty in data:
45 print("plotting duty cycle duration vs. time (" + str(targetDuty) + " CPU cycles)...")
46 circuitNum = len(data[targetDuty])
47 title = "$t_{duty}$ Duration Error vs. Time"
48 fig, axes = plt.subplots(nrows = circuitNum, ncols = 2, figsize = (13, 10))
49 fig.suptitle(title)
50 circuitIdx = 0
51 for circuit in data[targetDuty]:
52 for state in data[targetDuty][circuit]:
53 timestamps = np.array(data[targetDuty][circuit][state]["timestamps"]) / 1000000
54 dutys = np.array(data[targetDuty][circuit][state]["dutys"]) - targetDuty
55 pauses = np.array(data[targetDuty][circuit][state]["pauses"])
56 if (len(dutys) == 0) or (len(timestamps) == 0):
57 continue
58 dataLabel = "CO " + str(circuit + 1)
59 if (state == 0):
60 dataLabel += ", HIGH $t_{duty}$"
61 else:
62 dataLabel += ", LOW $t_{duty}$"
63 plots = None
64 if (circuitNum == 1):
65 plots = axes[state]
66 else:
67 plots = axes[circuitIdx, state]
68 plots.set_title(dataLabel)
69 for i in range(len(timestamps)):
70 if (pauses[i] == 1):
71 plots.axvline(x = timestamps[i], color = "r", linewidth = 5.0, alpha = 0.3)
72 plots.plot(timestamps, dutys, linewidth = 0.5, label = dataLabel)
73 plots.axhline(y = -500, color = "red", linestyle = "dashed", linewidth = 0.5)
74 plots.axhline(y = 500, color = "red", linestyle = "dashed", linewidth = 0.5)
75 plots.set_xlim(min(timestamps), max(timestamps))
76 plots.set_ylim(-1000, 1000)
77 plots.grid(which = "both")
78 circuitIdx += 1
79 fig.supxlabel("Time (s)")
80 fig.supylabel("$t_{duty}$ Error (CPU cycles)")
81 plt.tight_layout()
82 if (len(savePath) > 0):
83 fileDirPath = os.path.dirname(savePath) + "\\plots\\duration_v_time\\" + str(targetDuty) + "\\"
84 fileName = os.path.basename(savePath).replace(".txt", "")
85 fullSavePath = fileDirPath + fileName + ".png"
86 os.makedirs(fileDirPath, exist_ok = True)
87 plt.savefig(fullSavePath)
88 if (show):
89 plt.show()
90 plt.clf()
91 plt.close()
